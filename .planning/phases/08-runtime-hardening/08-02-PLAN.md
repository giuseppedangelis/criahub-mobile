    1. Introduzir um helper local dentro do mesmo `useEffect`, antes da definição de `bootstrap`:
       ```tsx
       const BOOTSTRAP_TIMEOUT_MS = 15000;

       const timeoutPromise = new Promise<never>((_, reject) => {
         const id = setTimeout(() => {
           reject(new Error("AuthContext bootstrap timeout (15s)"));
       ```

       Alternativa mais limpa (preferir se Promise.race não suportar cancel trivial): declarar `let timeoutId: ReturnType<typeof setTimeout> | undefined;` acima, criar `new Promise((_, reject) => { timeoutId = setTimeout(() => reject(...), 15000); })`, e chamar `clearTimeout(timeoutId)` no `finally` do bootstrap. Use esta forma. Código concreto:

       ```tsx
       useEffect(() => {
         let mounted = true;
         let timeoutId: ReturnType<typeof setTimeout> | undefined;

         const bootstrap = async () => {
           const BOOTSTRAP_TIMEOUT_MS = 15000;

           const timeoutPromise = new Promise<never>((_, reject) => {
             timeoutId = setTimeout(() => {
               reject(new Error("AuthContext bootstrap timeout (15s)"));
             }, BOOTSTRAP_TIMEOUT_MS);
           });

           const work = (async () => {
             const { data, error } = await supabase.auth.getSession();
             if (error) throw new Error(error.message);
             if (!mounted) return;
             const nextSession = data.session;
             setSession(nextSession);
             setUser(nextSession?.user ?? null);
             if (nextSession?.user) {
               await loadProfile(nextSession.user.id);
             } else {
               clearUserState();
             }
           })();

           try {
             // RUNTIME-02 (D-04): corta em 15s se Supabase estiver inacessível.
             // Reviewer da Apple testa em rede IPv6-only / VPN — spinner infinito vira
             // rejeição 2.1 "crash on launch / stuck on loading screen".
             await Promise.race([work, timeoutPromise]);
           } catch {
             // D-04: qualquer falha (incluindo timeout) → limpar estado e cair no login.
             if (mounted) {
               clearUserState();
             }
           } finally {
             // D-05: sempre libera spinner. Cancela timeout pendente caso work tenha vencido.
             if (timeoutId !== undefined) {
               clearTimeout(timeoutId);
             }
             if (mounted) {
               setIsLoading(false);
             }
           }
         };

         bootstrap();

         // ... onAuthStateChange handler existente segue INTACTO ...
       ```
